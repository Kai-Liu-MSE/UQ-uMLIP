from ase import Atoms
import os
import pandas as pd
import numpy as np
from tqdm import tqdm  # Import tqdm for the progress bar

from sevenn.calculator import SevenNetCalculator
# The 'modal' argument is required if the model is trained with multi-fidelity learning enabled.
calc = SevenNetCalculator(model='7net-mf-ompa', modal='mpa')

# Define the directory containing the .pckl.gzip files
input_folder = "../"

# Ensure the output folder exists

# List all .pckl.gzip files in the input folder
input_files = [f for f in os.listdir(input_folder) if f.endswith('.pckl.gzip')]

# Process each file
for file_name in tqdm(input_files, desc="Processing files"):
    # Construct the full file path
    file_path = os.path.join(input_folder, file_name)
    
    # Load the data
    df1 = pd.read_pickle(file_path, compression="gzip")
    
    atom1 = df1["ase_atoms"]
    pe1_pred = []
    force1_pred = []
    N1 = []
    structures = []
    
    for iatom in tqdm(atom1, desc="Processing atoms in file"):
        natom = iatom.get_global_number_of_atoms()
        iatom.calc = calc
        i_pe = iatom.get_potential_energy() / natom
        i_force = iatom.get_forces()
        iatom.calc = []
        structures.append({'ase_atoms': iatom, 'energy': i_pe, 'forces': i_force})
    
    # Create a DataFrame from the processed structures
    df = pd.DataFrame(structures)
    
    # Add energy corrected if needed
    reference_energy = 0
    df['energy_corrected'] = df['energy'] - reference_energy
    
    # Save the processed DataFrame
    output_file_name = os.path.splitext(file_name)[0] + '_new.pckl.gzip'
    df.to_pickle(output_file_name, compression='gzip', protocol=4)

print("All files processed and saved.")
